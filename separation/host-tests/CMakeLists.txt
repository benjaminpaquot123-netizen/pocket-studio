cmake_minimum_required(VERSION 3.22)
project(separation_host_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
# ORT_DIR: an unpacked onnxruntime-linux-x64 release (include/, lib/).
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../src/main/cpp)
set(DEMUCS ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/demucs.onnx/src)
set(ORT_INCLUDE ${CMAKE_CURRENT_BINARY_DIR}/ort-include)
file(GLOB ORT_HEADERS ${ORT_DIR}/include/*.h)
file(COPY ${ORT_HEADERS} DESTINATION ${ORT_INCLUDE}/onnxruntime/core/session)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../engine/third_party/libsamplerate.cmake)
add_executable(separation_test separation_test.cpp ${SRC}/Separator.cpp ${LIBSAMPLERATE_SOURCES} ${DEMUCS}/dsp.cpp ${DEMUCS}/model_apply.cpp ${DEMUCS}/model_inference.cpp)
target_include_directories(separation_test PRIVATE ${SRC} ${DEMUCS} ${ORT_INCLUDE} ${LIBSAMPLERATE_INCLUDES})
target_include_directories(separation_test SYSTEM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/eigen)
target_compile_options(separation_test PRIVATE -O2 -w)
target_link_directories(separation_test PRIVATE ${ORT_DIR}/lib)
target_link_libraries(separation_test PRIVATE onnxruntime)
set_target_properties(separation_test PROPERTIES BUILD_RPATH ${ORT_DIR}/lib)

# Basic Pitch transcription: needs only ORT and the model asset shipped in the app.
add_executable(pitch_test pitch_test.cpp ${SRC}/PitchTranscriber.cpp ${LIBSAMPLERATE_SOURCES})
target_include_directories(pitch_test PRIVATE ${SRC} ${ORT_DIR}/include ${LIBSAMPLERATE_INCLUDES})
target_compile_options(pitch_test PRIVATE -O2 -Wall -Wextra)
target_link_directories(pitch_test PRIVATE ${ORT_DIR}/lib)
target_link_libraries(pitch_test PRIVATE onnxruntime)
set_target_properties(pitch_test PROPERTIES BUILD_RPATH ${ORT_DIR}/lib)
